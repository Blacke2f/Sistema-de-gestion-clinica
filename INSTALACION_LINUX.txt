1. INSTALAR EN UBUNTU / DEBIAN / LINUX MINT
=====================================================================

  PASO 1 - Actualizar el sistema
  ----------------------------------------
  sudo apt update && sudo apt upgrade -y

  PASO 2 - Instalar Apache
  ----------------------------------------
  sudo apt install -y apache2
  sudo systemctl enable apache2
  sudo systemctl start apache2
  sudo a2enmod rewrite
  sudo a2enmod headers
  sudo systemctl restart apache2

  PASO 3 - Instalar PHP 8.3
  ----------------------------------------
  # Agregar repositorio Ondrej (PHP moderno para Debian/Ubuntu)
  sudo apt install -y software-properties-common curl gnupg
  sudo add-apt-repository ppa:ondrej/php -y   # Solo Ubuntu/Mint
  # Para Debian puro usar:
  # curl -sSLo /usr/share/keyrings/deb.sury.org-php.gpg \
  #   https://packages.sury.org/php/apt.gpg
  # echo "deb [signed-by=/usr/share/keyrings/deb.sury.org-php.gpg] \
  #   https://packages.sury.org/php/ $(lsb_release -sc) main" | \
  #   sudo tee /etc/apt/sources.list.d/php.list
  sudo apt update

  sudo apt install -y \
    php8.3 \
    php8.3-cli \
    php8.3-fpm \
    php8.3-mysql \
    php8.3-mbstring \
    php8.3-xml \
    php8.3-curl \
    php8.3-gd \
    php8.3-zip \
    php8.3-bcmath \
    php8.3-intl \
    php8.3-fileinfo \
    php8.3-opcache \
    libapache2-mod-php8.3

  # Verificar version instalada
  php -v

  PASO 4 - Instalar MariaDB
  ----------------------------------------
  sudo apt install -y mariadb-server mariadb-client
  sudo systemctl enable mariadb
  sudo systemctl start mariadb

  # Asegurar la instalacion
  sudo mysql_secure_installation
  # Responde:
  #   Enter password for root: [Enter si no hay]
  #   Switch to unix_socket authentication: N
  #   Change the root password: Y  -> ingresa una clave segura
  #   Remove anonymous users: Y
  #   Disallow root login remotely: Y
  #   Remove test database: Y
  #   Reload privilege tables: Y

  PASO 5 - Instalar Composer
  ----------------------------------------
  curl -sS https://getcomposer.org/installer | php
  sudo mv composer.phar /usr/local/bin/composer
  sudo chmod +x /usr/local/bin/composer
  composer --version

  PASO 6 - Copiar el proyecto
  ----------------------------------------
  # Opcion A: desde un archivo ZIP
  sudo mkdir -p /var/www/html/SISGECLIN
  sudo cp -r /ruta/al/SISGECLIN/* /var/www/html/SISGECLIN/

  # Opcion B: desde un pendrive montado
  sudo cp -r /media/usuario/USB/SISGECLIN /var/www/html/SISGECLIN

  # Permisos correctos para Apache
  sudo chown -R www-data:www-data /var/www/html/SISGECLIN
  sudo chmod -R 755 /var/www/html/SISGECLIN
  sudo chmod -R 775 /var/www/html/SISGECLIN/storage
  sudo chmod -R 775 /var/www/html/SISGECLIN/bootstrap/cache

  PASO 7 - Instalar dependencias del proyecto
  ----------------------------------------
  cd /var/www/html/SISGECLIN
  sudo -u www-data composer install --no-dev --optimize-autoloader

2. INSTALAR EN CENTOS / ROCKY LINUX / ALMALINUX
=====================================================================

  PASO 1 - Actualizar e instalar EPEL + Remi
  ----------------------------------------
  sudo dnf update -y
  sudo dnf install -y epel-release

  # Para Rocky/Alma/CentOS Stream 8:
  sudo dnf install -y https://rpms.remirepo.net/enterprise/remi-release-8.rpm
  sudo dnf module enable php:remi-8.3 -y

  # Para Rocky/Alma/CentOS Stream 9:
  sudo dnf install -y https://rpms.remirepo.net/enterprise/remi-release-9.rpm
  sudo dnf module enable php:remi-8.3 -y

  PASO 2 - Instalar PHP 8.3 y extensiones
  ----------------------------------------
  sudo dnf install -y \
    php \
    php-cli \
    php-fpm \
    php-mysqlnd \
    php-mbstring \
    php-xml \
    php-curl \
    php-gd \
    php-zip \
    php-bcmath \
    php-intl \
    php-fileinfo \
    php-opcache \
    php-json

  # Verificar
  php -v

  PASO 3 - Instalar Apache (httpd)
  ----------------------------------------
  sudo dnf install -y httpd
  sudo systemctl enable httpd
  sudo systemctl start httpd

  # Habilitar mod_rewrite
  echo 'LoadModule rewrite_module modules/mod_rewrite.so' | \
    sudo tee -a /etc/httpd/conf/httpd.conf
  echo '<Directory /var/www/html>' | sudo tee -a /etc/httpd/conf/httpd.conf
  echo '  AllowOverride All' | sudo tee -a /etc/httpd/conf/httpd.conf
  echo '</Directory>' | sudo tee -a /etc/httpd/conf/httpd.conf

  PASO 4 - Instalar MariaDB
  ----------------------------------------
  sudo dnf install -y mariadb-server
  sudo systemctl enable mariadb
  sudo systemctl start mariadb
  sudo mysql_secure_installation

  PASO 5 - Firewall del servidor
  ----------------------------------------
  sudo firewall-cmd --permanent --add-service=http
  sudo firewall-cmd --permanent --add-service=https
  sudo firewall-cmd --permanent --add-port=80/tcp
  sudo firewall-cmd --reload

  # Verificar reglas activas
  sudo firewall-cmd --list-all

  PASO 6 - SELinux (si esta habilitado)
  ----------------------------------------
  # Permitir que Apache acceda a la BD y al filesystem
  sudo setsebool -P httpd_can_network_connect_db 1
  sudo setsebool -P httpd_unified 1
  sudo restorecon -R /var/www/html/SISGECLIN

  # Verificar estado de SELinux
  getenforce

  PASO 7 - Instalar Composer y copiar proyecto
  ----------------------------------------
  # Igual que Ubuntu (Paso 5 y 6 de la seccion 1)
  curl -sS https://getcomposer.org/installer | php
  sudo mv composer.phar /usr/local/bin/composer

  sudo cp -r /ruta/SISGECLIN /var/www/html/SISGECLIN
  sudo chown -R apache:apache /var/www/html/SISGECLIN
  sudo chmod -R 775 /var/www/html/SISGECLIN/storage
  sudo chmod -R 775 /var/www/html/SISGECLIN/bootstrap/cache

  cd /var/www/html/SISGECLIN
  sudo -u apache composer install --no-dev --optimize-autoloader

4. INSTALAR CON DOCKER (CUALQUIER DISTRO)
=====================================================================

  Instalar Docker y Docker Compose:
  ----------------------------------------
  # Ubuntu/Debian
  curl -fsSL https://get.docker.com | sh
  sudo usermod -aG docker $USER
  sudo apt install -y docker-compose-plugin

  # CentOS/Rocky/Fedora
  sudo dnf install -y docker docker-compose
  sudo systemctl enable --now docker

  Crear docker-compose.yml en la raiz del proyecto:
  ----------------------------------------
  version: '3.8'

  services:
    app:
      image: php:8.3-apache
      container_name: sisgeclin_app
      ports:
        - "80:80"
      volumes:
        - .:/var/www/html
        - ./docker/apache.conf:/etc/apache2/sites-enabled/000-default.conf
      depends_on:
        - db
      environment:
        - APACHE_DOCUMENT_ROOT=/var/www/html/public

    db:
      image: mariadb:10.6
      container_name: sisgeclin_db
      environment:
        MYSQL_ROOT_PASSWORD: secret
        MYSQL_DATABASE: sisgeclin_laravel
        MYSQL_USER: sisgeclin
        MYSQL_PASSWORD: sisgeclin123
      volumes:
        - db_data:/var/lib/mysql
        - ./database/sisgeclin_laravel.sql:/docker-entrypoint-initdb.d/init.sql

    phpmyadmin:
      image: phpmyadmin/phpmyadmin
      ports:
        - "8080:80"
      environment:
        PMA_HOST: db

  volumes:
    db_data:

  Ejecutar:
  ----------------------------------------
  cd /ruta/a/SISGECLIN
  docker compose up -d
  docker compose exec app composer install
  docker compose exec app php artisan optimize:clear
  docker compose exec app php artisan storage:link

  Acceder: http://localhost

5. CONFIGURAR LA BASE DE DATOS
=====================================================================

  Crear la base de datos:
  ----------------------------------------
  sudo mysql -u root -p

  # En el prompt de MySQL/MariaDB:
  CREATE DATABASE sisgeclin_laravel
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;

  CREATE USER 'sisgeclin'@'localhost' IDENTIFIED BY 'ClaveSegura2026!';
  GRANT ALL PRIVILEGES ON sisgeclin_laravel.* TO 'sisgeclin'@'localhost';
  FLUSH PRIVILEGES;
  EXIT;

  Importar el dump SQL con datos de prueba:
  ----------------------------------------
  mysql -u sisgeclin -p sisgeclin_laravel \
    < /var/www/html/SISGECLIN/database/sisgeclin_laravel.sql

  O usar migraciones y seeders (BD limpia):
  ----------------------------------------
  cd /var/www/html/SISGECLIN
  php artisan migrate --force
  php artisan db:seed --class=DatosPruebaSeeder

6. CONFIGURAR EL PROYECTO (.env)
=====================================================================

  Copiar y editar el archivo de configuracion:
  ----------------------------------------
  cd /var/www/html/SISGECLIN
  cp .env.example .env
  nano .env

  Lineas a modificar:
  ----------------------------------------
  APP_NAME="SISGECLIN"
  APP_ENV=production
  APP_DEBUG=false
  APP_URL=http://[IP-DEL-SERVIDOR]/SISGECLIN/public

  DB_CONNECTION=mysql
  DB_HOST=127.0.0.1
  DB_PORT=3306
  DB_DATABASE=sisgeclin_laravel
  DB_USERNAME=sisgeclin
  DB_PASSWORD=ClaveSegura2026!

  SESSION_DRIVER=file
  SESSION_LIFETIME=480
  SESSION_DOMAIN=

  Aplicar configuracion y crear storage link:
  ----------------------------------------
  php artisan optimize:clear
  php artisan storage:link
  php artisan optimize

  Permisos finales:
  ----------------------------------------
  # Ubuntu/Debian
  sudo chown -R www-data:www-data storage bootstrap/cache
  sudo chmod -R 775 storage bootstrap/cache

  # CentOS/Rocky/Fedora
  sudo chown -R apache:apache storage bootstrap/cache
  sudo chmod -R 775 storage bootstrap/cache

7. CONFIGURAR APACHE VIRTUALHOST
=====================================================================

  Ubuntu/Debian:
  ----------------------------------------
  sudo nano /etc/apache2/sites-available/sisgeclin.conf

  Contenido del archivo:
  ----------------------------------------
  <VirtualHost *:80>
      ServerName sisgeclin.example.com
      # O usa la IP si no tienes dominio:
      # ServerName 192.168.1.10

      DocumentRoot /var/www/html/SISGECLIN/public

      <Directory /var/www/html/SISGECLIN/public>
          AllowOverride All
          Options -Indexes +FollowSymLinks
          Require all granted
      </Directory>

      # Logs
      ErrorLog ${APACHE_LOG_DIR}/sisgeclin_error.log
      CustomLog ${APACHE_LOG_DIR}/sisgeclin_access.log combined
  </VirtualHost>

  Activar el sitio:
  ----------------------------------------
  sudo a2ensite sisgeclin.conf
  sudo a2dissite 000-default.conf    # deshabilitar sitio por defecto
  sudo apache2ctl configtest         # verificar sintaxis
  sudo systemctl restart apache2

  CentOS/Rocky/AlmaLinux:
  ----------------------------------------
  sudo nano /etc/httpd/conf.d/sisgeclin.conf

  # Mismo contenido que arriba pero logs con ruta RHEL:
  ErrorLog /var/log/httpd/sisgeclin_error.log
  CustomLog /var/log/httpd/sisgeclin_access.log combined

  sudo httpd -t                      # verificar sintaxis
  sudo systemctl restart httpd

  Acceso sin VirtualHost (URL directa por IP):
  ----------------------------------------
  Si solo tienes una IP y prefieres no usar VirtualHost,
  el sistema tambien funciona accediendo directamente:
  http://[IP-SERVIDOR]/SISGECLIN/public

  Para esto, asegurate de que en la config de Apache exista:
  <Directory /var/www/html>
    AllowOverride All
    Require all granted
  </Directory>

8. ACCESO EN RED LOCAL
=====================================================================

  Obtener la IP del servidor Linux:
  ----------------------------------------
  ip addr show | grep "inet " | grep -v "127.0.0.1"
  # O mas simple:
  hostname -I | awk '{print $1}'

  Abrir el firewall (Linux):
  ----------------------------------------
  # Ubuntu/Debian con ufw:
  sudo ufw allow 80/tcp
  sudo ufw allow 22/tcp    # SSH - no cerrar
  sudo ufw enable
  sudo ufw status

  # CentOS/Rocky con firewalld:
  sudo firewall-cmd --permanent --add-service=http
  sudo firewall-cmd --permanent --add-port=80/tcp
  sudo firewall-cmd --reload
  sudo firewall-cmd --list-all

  # Verificar que Apache escucha en todas las interfaces:
  sudo netstat -tlnp | grep :80
  # Debe mostrar: 0.0.0.0:80

  Acceder desde otro dispositivo:
  ----------------------------------------
  En el navegador del telefono o tablet:
  http://192.168.1.10/SISGECLIN/public

  (Reemplaza 192.168.1.10 con tu IP real)

9. BACKUP AUTOMATICO (CRON)
=====================================================================

  Configurar el crontab:
  ----------------------------------------
  sudo crontab -e -u www-data     # Ubuntu/Debian
  # o
  sudo crontab -e -u apache       # CentOS/Rocky

  Agregar estas lineas al crontab:
  ----------------------------------------
  # Backup de BD a las 2:00 AM todos los dias
  0 2 * * * /usr/bin/php /var/www/html/SISGECLIN/artisan db:backup --keep=7

  # Ejecutar el scheduler de Laravel cada minuto
  * * * * * /usr/bin/php /var/www/html/SISGECLIN/artisan schedule:run >> /dev/null 2>&1

  Verificar que cron esta activo:
  ----------------------------------------
  # Ubuntu/Debian:
  sudo systemctl status cron

  # CentOS/Rocky/Fedora:
  sudo systemctl status crond

  Backup manual:
  ----------------------------------------
  cd /var/www/html/SISGECLIN
  php artisan db:backup
  ls storage/app/backups/

10. ACTUALIZACION DEL SISTEMA
=====================================================================

  Cuando recibas una nueva version de SISGECLIN:
  ----------------------------------------
  cd /var/www/html/SISGECLIN

  # 1. Hacer backup de la BD antes de actualizar
  php artisan db:backup

  # 2. Copiar los nuevos archivos (sin sobreescribir .env)
  # (copia todos excepto .env y storage/)

  # 3. Actualizar dependencias
  sudo -u www-data composer install --no-dev --optimize-autoloader

  # 4. Ejecutar migraciones nuevas
  php artisan migrate --force

  # 5. Limpiar caches
  php artisan optimize:clear
  php artisan optimize

  # 6. Reiniciar Apache
  sudo systemctl restart apache2     # Ubuntu
  sudo systemctl restart httpd       # CentOS

11. SOLUCION DE PROBLEMAS EN LINUX
=====================================================================

  Error 403 Forbidden
  ----------------------------------------
  1. Verifica AllowOverride All en la config de Apache
  2. Verifica permisos: ls -la /var/www/html/SISGECLIN
  3. sudo chmod -R 755 /var/www/html/SISGECLIN
  4. sudo chown -R www-data:www-data /var/www/html/SISGECLIN
  5. sudo systemctl restart apache2

  Error 500 (Internal Server Error)
  ----------------------------------------
  1. Activa debug temporalmente en .env: APP_DEBUG=true
  2. Revisa los logs de Apache:
     sudo tail -50 /var/log/apache2/sisgeclin_error.log   # Ubuntu
     sudo tail -50 /var/log/httpd/sisgeclin_error.log     # CentOS
  3. Revisa el log de Laravel:
     tail -50 storage/logs/laravel.log
  4. Ejecuta: php artisan optimize:clear

  PHP no encuentra extensiones
  ----------------------------------------
  # Ubuntu/Debian
  sudo apt install php8.3-gd php8.3-mbstring php8.3-zip php8.3-intl
  sudo systemctl restart apache2

  # CentOS/Rocky
  sudo dnf install php-gd php-mbstring php-zip php-intl
  sudo systemctl restart httpd

  # Verificar extensiones activas:
  php -m | grep -E "gd|mbstring|mysql|zip|opcache"

  Error de permisos en storage/
  ----------------------------------------
  sudo chown -R www-data:www-data /var/www/html/SISGECLIN/storage
  sudo chmod -R 775 /var/www/html/SISGECLIN/storage
  # Si persiste (SELinux):
  sudo chcon -R -t httpd_sys_rw_content_t \
    /var/www/html/SISGECLIN/storage

  MariaDB: acceso denegado
  ----------------------------------------
  sudo mysql -u root
  ALTER USER 'sisgeclin'@'localhost' IDENTIFIED BY 'NuevaClave';
  FLUSH PRIVILEGES;
  EXIT;

  mod_rewrite no funciona
  ----------------------------------------
  sudo a2enmod rewrite
  sudo apache2ctl -M | grep rewrite   # debe aparecer rewrite_module
  # Verificar que AllowOverride All este configurado
  sudo systemctl restart apache2

  Las URLs limpias no funcionan
  ----------------------------------------
  1. Verifica que .htaccess existe en public/
  2. Si no existe, recrea el .htaccess estandar de Laravel:

  Contenido de public/.htaccess:
  <IfModule mod_rewrite.c>
      <IfModule mod_negotiation.c>
          Options -MultiViews -Indexes
      </IfModule>
      RewriteEngine On
      RewriteCond %{HTTP:Authorization} .
      RewriteRule .* - [E=HTTP_AUTHORIZATION:%{HTTP:Authorization}]
      RewriteCond %{REQUEST_FILENAME} !-d
      RewriteCond %{REQUEST_FILENAME} !-f
      RewriteRule ^ index.php [L]
  </IfModule>

  SELinux bloquea Apache
  ----------------------------------------
  sudo setsebool -P httpd_can_network_connect_db 1
  sudo setsebool -P httpd_unified 1
  sudo ausearch -c 'httpd' --raw | audit2allow -M sisgeclin
  sudo semodule -i sisgeclin.pp
